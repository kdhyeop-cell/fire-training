// 💡 [최종 수정됨] 장소 검색 시 최상단 결과로 "강제 줌인(Zoom-in)" 하는 로직 function searchPlaces() { var rawKeyword = document.getElementById('keyword').value; if (!rawKeyword.trim()) return alert('검색어를 입력해주세요!'); // 제주도 우선 검색어 조합 var searchKeyword = rawKeyword.includes('제주') ? rawKeyword : '제주 ' + rawKeyword; document.getElementById('road-info-card').classList.add('hidden'); ps.keywordSearch(searchKeyword, function(data, status) { if (status === kakao.maps.services.Status.OK) { // 💡 [변경] Bounds(영역 넓히기) 기능을 삭제하고, 무조건 첫 번째 결과로 중심 이동 후 레벨 3으로 강제 줌인! var firstResultPosition = new kakao.maps.LatLng(data[0].y, data[0].x); map.setCenter(firstResultPosition); map.setLevel(3); // 1~14 중 3(확대) 레벨 고정 checkRoadWidth(rawKeyword, data[0].address_name); } else { geocoder.addressSearch(searchKeyword, function(result, status) { if (status === kakao.maps.services.Status.OK) { var addressPosition = new kakao.maps.LatLng(result[0].y, result[0].x); map.setCenter(addressPosition); map.setLevel(3); // 주소 검색 시에도 3 레벨 고정 checkRoadWidth(rawKeyword, result[0].address_name); } else { alert('제주도 내에서 "' + rawKeyword + '" 검색 결과를 찾을 수 없습니다.'); } }); } }); }
